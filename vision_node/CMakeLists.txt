cmake_minimum_required(VERSION 3.16)
project(hoodseek_vision LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(HOODSEEK_ECOAI "Build the ATLAS EcoAI camera/image runner" OFF)
find_package(PkgConfig REQUIRED)
# ATLAS SDK's OpenCV CMake export references absent optional 3rd-party archives.
# pkg-config provides the installed libraries without importing those targets.
pkg_check_modules(OPENCV REQUIRED opencv4)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)
add_library(vision_core src/vision.cpp)
target_include_directories(vision_core PUBLIC src)
target_include_directories(vision_core SYSTEM PUBLIC ${OPENCV_INCLUDE_DIRS})
target_link_directories(vision_core PUBLIC ${OPENCV_LIBRARY_DIRS})
target_compile_options(vision_core PUBLIC ${OPENCV_CFLAGS_OTHER})
# Only ship dependencies used by this observer; opencv4.pc also lists optional modules.
target_link_libraries(vision_core PUBLIC opencv_imgproc opencv_core nlohmann_json::nlohmann_json)
if(NOT HOODSEEK_ECOAI)
  # The ATLAS device image only ships opencv-core/imgproc (no videoio/imgcodecs
  # package); the ECOAI build reads the camera via V4L2 directly instead.
  # imgcodecs is only for the PC-side --export-input path.
  target_link_libraries(vision_core PUBLIC opencv_imgcodecs)
endif()
target_compile_options(vision_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(hoodseek_vision src/main.cpp src/link.cpp)
target_link_libraries(hoodseek_vision PRIVATE vision_core Threads::Threads)
target_compile_options(hoodseek_vision PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(HOODSEEK_ECOAI)
  pkg_check_modules(ECOAI REQUIRED IMPORTED_TARGET ecoai-inference)
  target_sources(hoodseek_vision PRIVATE src/ecoai_detector.cpp src/v4l2_camera.cpp src/preview.cpp)
  # dlopen: the camera-card video loads libturbojpeg.so.0 at runtime (no header or link library in the SDK).
  target_link_libraries(hoodseek_vision PRIVATE PkgConfig::ECOAI ${CMAKE_DL_LIBS})
  target_compile_definitions(hoodseek_vision PRIVATE HOODSEEK_ECOAI=1)
endif()
include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_executable(vision_tests tests/vision_tests.cpp)
  target_link_libraries(vision_tests PRIVATE vision_core Threads::Threads)
  target_compile_options(vision_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME vision_core COMMAND vision_tests)
  add_executable(preview_tests tests/preview_tests.cpp src/preview.cpp)
  target_link_libraries(preview_tests PRIVATE vision_core Threads::Threads)
  target_compile_options(preview_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME vision_preview COMMAND preview_tests)
  set_tests_properties(vision_preview PROPERTIES TIMEOUT 20)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME vision_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py $<TARGET_FILE:hoodseek_vision>)
  # PC-side #49 analysis tool; needs python3 cv2 + numpy.
  add_test(NAME vision_table COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/detection_table_test.py)
endif()
