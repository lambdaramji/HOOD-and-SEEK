cmake_minimum_required(VERSION 3.20)
project(hoodseek_voice_parser LANGUAGES CXX)

add_library(hoodseek_voice_parser src/command_parser.cpp)
target_include_directories(hoodseek_voice_parser PUBLIC src)
target_compile_features(hoodseek_voice_parser PUBLIC cxx_std_17)
target_compile_options(hoodseek_voice_parser PRIVATE -Wall -Wextra -Wpedantic -Werror)

add_library(hoodseek_voice_audio src/audio_pipeline.cpp)
target_include_directories(hoodseek_voice_audio PUBLIC src)
target_compile_features(hoodseek_voice_audio PUBLIC cxx_std_17)
target_compile_options(hoodseek_voice_audio PRIVATE -Wall -Wextra -Wpedantic -Werror)

add_library(hoodseek_voice_tcp src/voice_tcp.cpp)
target_include_directories(hoodseek_voice_tcp PUBLIC src)
find_package(Threads REQUIRED)
target_link_libraries(hoodseek_voice_tcp PUBLIC hoodseek_voice_parser Threads::Threads)
target_compile_options(hoodseek_voice_tcp PRIVATE -Wall -Wextra -Wpedantic -Werror)

set(WHISPER_SOURCE_DIR "" CACHE PATH "Existing pinned whisper.cpp source from issue 50")
set(SHERPA_ONNX_ROOT "" CACHE PATH "Existing sherpa-onnx CPU distribution for optional SenseVoice")
option(HOOD_SEEK_WITH_PULSE "Build existing GStreamer PulseAudio capture" OFF)
if(WHISPER_SOURCE_DIR)
  set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_EXAMPLES ON CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
  set(WHISPER_COMMON_FFMPEG OFF CACHE BOOL "" FORCE)
  add_subdirectory("${WHISPER_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/whisper" EXCLUDE_FROM_ALL)
  add_executable(hoodseek-voice-observer src/audio_observer.cpp)
  target_include_directories(hoodseek-voice-observer PRIVATE "${WHISPER_SOURCE_DIR}/examples")
  target_link_libraries(hoodseek-voice-observer PRIVATE hoodseek_voice_tcp hoodseek_voice_audio whisper common)
  target_compile_options(hoodseek-voice-observer PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(SHERPA_ONNX_ROOT)
    find_path(SHERPA_ONNX_INCLUDE_DIR sherpa-onnx/c-api/c-api.h
      PATHS "${SHERPA_ONNX_ROOT}/include" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    find_library(SHERPA_ONNX_C_API NAMES sherpa-onnx-c-api
      PATHS "${SHERPA_ONNX_ROOT}/lib" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
    target_include_directories(hoodseek-voice-observer PRIVATE "${SHERPA_ONNX_INCLUDE_DIR}")
    target_link_libraries(hoodseek-voice-observer PRIVATE "${SHERPA_ONNX_C_API}")
    target_compile_definitions(hoodseek-voice-observer PRIVATE HOOD_SEEK_WITH_SENSEVOICE=1)
    set_property(TARGET hoodseek-voice-observer APPEND PROPERTY BUILD_RPATH "${SHERPA_ONNX_ROOT}/lib")
  endif()
  if(BUILD_TESTING)
    add_test(NAME voice-asr-options COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/voice_asr_options_test.py" "$<TARGET_FILE:hoodseek-voice-observer>")
  endif()
  # 원음·모델을 명시하는 오프라인 점수 대조 도구. 기본 빌드·제품 실행 경로에는 넣지 않는다.

  target_compile_features(voice_candidate_probe PRIVATE cxx_std_17)

    pkg_check_modules(GSTREAMER REQUIRED IMPORTED_TARGET gstreamer-app-1.0)
    pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse)
    target_link_libraries(hoodseek-voice-observer PRIVATE PkgConfig::GSTREAMER PkgConfig::PULSE)
    target_compile_definitions(hoodseek-voice-observer PRIVATE HOOD_SEEK_WITH_PULSE=1)
  endif()
endif()
